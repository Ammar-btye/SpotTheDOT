**Shared start-server helper.** Paste it once per shell:
```bash
start_llama () {  # $1 = gguf path, $2 = extra flags
  pkill -f llama-server; sleep 1
  nohup ~/llama.cpp/build/bin/llama-server -m "$1" $2 -ngl all -fa on -lm none -c 16384 -np 1 -rea off \
    --host 127.0.0.1 --port 8080 --metrics > ~/llama-server.log 2>&1 &
  for i in $(seq 300); do curl -s 127.0.0.1:8080/health | grep -q ok && { echo READY; return 0; }; \
    pgrep -f llama-server >/dev/null || break; sleep 2; done
  echo "NOT READY: last log lines:"; tail -20 ~/llama-server.log; return 1   # e.g. an unknown flag on an older build
}
```

**Push helper.** If `git push` is refused on a box, use the `tar` fallback in `amd/system2/results/report/README.md`
(e).
```bash
push_results () {  # $1 = branch name, then paths

  git commit -m "results: $b (run on $(hostname), $(date -u +%FT%TZ))" && git push -u origin "$b"
}
```

## GPU D: vision readiness probe (fits the 16 GB environment)

Decides whether the VLM can look at images at all on this stack (`research/tech-spot-vision-cv-2026-10-10.md` §5).
```bash
hf download unsloth/Qwen3.5-9B-GGUF Qwen3.5-9B-Q4_K_M.gguf mmproj-F16.gguf --local-dir ~/models/q35
start_llama ~/models/q35/Qwen3.5-9B-Q4_K_M.gguf "--mmproj $HOME/models/q35/mmproj-F16.gguf --image-max-tokens 1024"
cd ~/SpotTheDOT && cat > /tmp/vprobe.py <<'EOF'
import base64, json, sys, time, urllib.request
K = 'SPOTDATA/SPOT_KnightHacks_Challenge_Kit Part One/04_ASSETS/sample_inspection_images/'
schema = {"type": "object", "additionalProperties": False, "required": ["component", "hot", "scale_max_c"],
          "properties": {"component": {"type": "string", "maxLength": 60}, "hot": {"type": "boolean"},
                         "scale_max_c": {"type": "integer"}}}
def ask(img, text, fmt=None):
    b64 = base64.b64encode(open(K + img, 'rb').read()).decode()
    body = {"messages": [{"role": "user", "content": [
        {"type": "image_url", "image_url": {"url": "data:image/jpeg;base64," + b64}}, {"type": "text", "text": text}]}],
        "temperature": 0, "max_tokens": 120, "chat_template_kwargs": {"enable_thinking": False}}
    if fmt: body["response_format"] = {"type": "json_schema", "json_schema": {"name": "probe", "schema": fmt}}
    t = time.time(); req = urllib.request.Request("http://127.0.0.1:8080/v1/chat/completions", json.dumps(body).encode(),
                                                  {"Content-Type": "application/json"})
    out = json.load(urllib.request.urlopen(req, timeout=120))["choices"][0]["message"]["content"]
    return round(time.time() - t, 2), out
print("T1 image+schema:", ask("thermal_pump_bearing_01.jpg", "Which component is hottest, and what is the colour bar's top value?", schema))
print("T2 gauge free text:", ask("gauge_pressure_01.jpg", "List the printed scale labels and the unit. Do not estimate the reading."))
a = ask("visual_pump_motor_01.jpg", "Describe the image in one sentence.")
b = ask("valve_detail_01.jpg", "Describe the image in one sentence.")
print("T3 leak check (2nd answer must be about a valve, not a pump motor):", a, b)
EOF
python3 /tmp/vprobe.py; grep -iE "error|fault|abort" ~/llama-server.log | tail -5
```
Pass criteria:
- **T1:** returns schema-valid JSON naming the bearing or motor end, with a scale top of 100.
- **T2:** lists the labels, including "290", plus psi.
- **T3:** the second answer describes the valve, so no content leaks across requests (llama.cpp #28056).

If T1 fails only because of the schema, the fallback is two calls (free text, then schema). If outputs look
corrupted, retry with `-fa off` (llama.cpp #29629).

Paste the full output; there are no files to push.
